cmake_minimum_required(VERSION 3.16)
project(evtol_vision CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio calib3d objdetect)
find_package(yaml-cpp REQUIRED)
add_library(evtol_detector STATIC src/detector.cpp src/frame_sources.cpp)
target_include_directories(evtol_detector PUBLIC include src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(evtol_detector PUBLIC ${OpenCV_LIBS} yaml-cpp)
add_executable(vision_node src/main.cpp)
target_link_libraries(vision_node evtol_detector)
add_executable(parity_tool src/parity_tool.cpp)
target_link_libraries(parity_tool evtol_detector)
